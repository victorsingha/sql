Delete Duplicate Records


WITH CTE AS (
    SELECT *,
           ROW_NUMBER() OVER (PARTITION BY COL1, COL2 ORDER BY (SELECT NULL)) AS rn
    FROM YOUR_TABLE_NAME
)
DELETE FROM CTE WHERE rn > 1;
